cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(euno_native_probes)
target_sources(app PRIVATE main.c)
target_compile_definitions(app PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(app PRIVATE ../common)
target_link_libraries(zephyr_interface INTERFACE -Wl,--wrap=z_cstart)
if(EUNO_MATRIX)
 target_sources(app PRIVATE matrix_probes.c)
 target_compile_definitions(app PRIVATE EUNO_MATRIX=1)
 target_link_libraries(zephyr_interface INTERFACE -Wl,--wrap=sys_clock_announce)
endif()
